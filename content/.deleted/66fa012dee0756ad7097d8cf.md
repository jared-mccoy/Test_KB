const regex: {
	footnoteBody: [
		`(?!^\d+\.\s)(?<!\d|\[\^)(\d+)(?!\.\d|\))`
		`[^$1]`
	]
	footnoteBodyEpub: [
		`\n\n\[\]\{#(\d+)_.*?\.anchor\}\s*\*?\[\^\1\]\*?\s.*?(\r?\n)?`,
		`\[\^\[(\d+)\]\{\.underline\}\^\]\(#\d+_\d+\)` // alternate		
		`\n[^$1]: `
	],
	paginationBreak:[
		`\n\s*(\d+)\s*\n`
		` `
	]
	promoteParToSec6:[
		`^\s*(\d+)(\\?\.)\s+(.*)`
		`###### $1\n$3`
	]
}
